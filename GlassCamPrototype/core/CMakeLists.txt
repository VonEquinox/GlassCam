cmake_minimum_required(VERSION 3.16)
project(GlassCamHNCS LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(glasscam_hncs STATIC hncs.cpp)
target_link_libraries(glasscam_hncs PUBLIC Threads::Threads)
target_compile_features(glasscam_hncs PUBLIC cxx_std_17)
target_include_directories(glasscam_hncs PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(glasscam_hncs PRIVATE -O3 -fno-fast-math -ffp-contract=off -Wall -Wextra -Wpedantic)
elseif(MSVC)
  target_compile_options(glasscam_hncs PRIVATE /O2 /fp:strict)
endif()
add_executable(hncs_cli hncs_cli.cpp)
target_link_libraries(hncs_cli PRIVATE glasscam_hncs)
enable_testing()
add_executable(hncs_smoke hncs_smoke.cpp)
target_link_libraries(hncs_smoke PRIVATE glasscam_hncs)
add_test(NAME hncs_smoke COMMAND hncs_smoke)
