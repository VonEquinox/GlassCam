cmake_minimum_required(VERSION 3.16)
project(GlassCamHNCS LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(glasscam_hncs STATIC hncs.cpp hncs_spatial.cpp hncs_numpy_rng.cpp)
target_link_libraries(glasscam_hncs PUBLIC Threads::Threads)
target_compile_features(glasscam_hncs PUBLIC cxx_std_17)
target_include_directories(glasscam_hncs PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(glasscam_hncs PRIVATE -O3 -fno-fast-math -ffp-contract=off -Wall -Wextra -Wpedantic)
elseif(MSVC)
  target_compile_options(glasscam_hncs PRIVATE /O2 /fp:strict)
endif()
add_executable(hncs_cli hncs_cli.cpp)
target_link_libraries(hncs_cli PRIVATE glasscam_hncs)
enable_testing()
add_executable(hncs_smoke hncs_smoke.cpp)
target_link_libraries(hncs_smoke PRIVATE glasscam_hncs)
add_test(NAME hncs_smoke COMMAND hncs_smoke)
add_executable(hncs_spatial_smoke ../tests/native-legacy/hncs_spatial_smoke.cpp)
target_link_libraries(hncs_spatial_smoke PRIVATE glasscam_hncs)
add_test(NAME hncs_spatial_smoke COMMAND hncs_spatial_smoke)
include(ffcs.cmake)
add_library(glasscam_f5 STATIC f5_math.cpp f5_registration.cpp f5_quality.cpp f5_c.cpp)
target_compile_features(glasscam_f5 PUBLIC cxx_std_17)
target_include_directories(glasscam_f5 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(glasscam_f5 PRIVATE -O2 -fno-fast-math -ffp-contract=off -Wall -Wextra)
endif()
add_executable(f5_core_tests ../tests/native-legacy/f5_core_tests.cpp)
target_link_libraries(f5_core_tests PRIVATE glasscam_f5)
add_test(NAME f5_core_contracts COMMAND f5_core_tests)
add_library(glasscam_lumix STATIC lumix.cpp)
target_link_libraries(glasscam_lumix PUBLIC ZLIB::ZLIB)
target_compile_features(glasscam_lumix PUBLIC cxx_std_17)
target_include_directories(glasscam_lumix PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(glasscam_lumix PRIVATE -O3 -fno-fast-math -ffp-contract=off -Wall -Wextra -Wpedantic)
endif()
